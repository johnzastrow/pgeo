% Report styling (pandoc + XeLaTeX). Sans-serif throughout; compact tables; figures sized to
% the text width; no automatic figure numbering (the report numbers figures itself).
\usepackage{etoolbox}
% Bibliography entries are largely URLs. xurl lets \url{} break at any character, so a long
% reference wraps instead of running into the margin.
\usepackage{xurl}
\renewcommand{\familydefault}{\sfdefault}
% Narrow column padding: the widest tables have nine columns, and pandoc computes each column from
% (\linewidth - 2n\tabcolsep), so a smaller separation gives the cells the room back.
% Grid lines are added to the generated LaTeX (report/pdf/gridtables.py); these give the
% cells room to breathe inside them: taller rows, and a little more space either side.
\AtBeginEnvironment{longtable}{\footnotesize\setlength{\tabcolsep}{4pt}\renewcommand{\arraystretch}{1.35}}
\setlength{\arrayrulewidth}{0.3pt}
\usepackage[table]{xcolor}
\arrayrulecolor{black!35}
% Hyphenate the first word of a table cell. TeX never hyphenates a word that starts a paragraph
% with no preceding glue, so a long single word ("autocomplete") in a narrow column ran past the
% cell edge; a zero-width space in front of it restores hyphenation.
\makeatletter
\let\pandoc@raggedright\raggedright
\renewcommand{\raggedright}{\pandoc@raggedright\hspace{0pt}}
\let\pandoc@raggedleft\raggedleft
\renewcommand{\raggedleft}{\pandoc@raggedleft\hspace{0pt}}
\makeatother
\setlength{\LTleft}{0pt}
\setlength{\LTright}{0pt}
% Keep a table's caption with the table: without this a page break can leave the caption
% stranded at the foot of the previous page.
\usepackage{needspace}
\usepackage{float}
\floatplacement{figure}{H}
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth,keepaspectratio}
\usepackage{caption}
\captionsetup{labelformat=empty}
\usepackage{titlesec}
\titleformat{\section}{\Large\bfseries\color[HTML]{1f5f8b}}{}{0pt}{}
\titleformat{\subsection}{\large\bfseries}{}{0pt}{}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small pgeo: a geocoder inside PostgreSQL}
\fancyhead[R]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\setlength{\headheight}{14pt}

% Callout boxes for the statements that matter most (renderer emits \begin{keybox} etc.)
\usepackage{tcolorbox}
\tcbuselibrary{skins,breakable}

  boxrule=0.6pt, left=6pt, right=6pt, top=5pt, bottom=5pt, title=Key result,
  fonttitle=\bfseries\small, coltitle=white}
\newtcolorbox{impactbox}{breakable, enhanced, colback=green!4!white, colframe=green!35!black,
  boxrule=0.6pt, left=6pt, right=6pt, top=5pt, bottom=5pt, title=What this means,
  fonttitle=\bfseries\small, coltitle=white}
\newtcolorbox{cautionbox}{breakable, enhanced, colback=orange!6!white, colframe=orange!65!black,
  boxrule=0.6pt, left=6pt, right=6pt, top=5pt, bottom=5pt, title=Caution,
  fonttitle=\bfseries\small, coltitle=white}
